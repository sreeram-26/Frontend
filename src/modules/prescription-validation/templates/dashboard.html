Triggers & Stored Procedures
TRIGGER: Auto-Audit on New Prescription
TRIGGER after_insert_prescription AFTER INSERT ON prescriptions FOR EACH ROW BEGIN CALL run_prescription_audit(NEW.prescription_id); IF compliance_score < 75 THEN UPDATE prescriptions SET status = 'flagged' WHERE prescription_id = NEW.prescription_id; END IF; INSERT INTO audit_logs (action, prescription_id, timestamp) VALUES ('AUTO_AUDIT', NEW.prescription_id, NOW()); END;
STORED PROCEDURE: run_prescription_audit(rx_id)
PROCEDURE run_prescription_audit(IN rx_id VARCHAR(20)) BEGIN FOR EACH item IN prescription_items WHERE prescription_id = rx_id DO IF item.dosage_amount > drugs.max_dose THEN INSERT INTO violations VALUES ('DOSAGE_EXCEEDED','HIGH',rx_id); END IF; IF item.frequency NOT IN allowed_frequencies THEN INSERT INTO violations VALUES ('INVALID_FREQUENCY','MEDIUM',rx_id); END IF; IF item.route NOT IN allowed_routes THEN INSERT INTO violations VALUES ('INVALID_ROUTE','MEDIUM',rx_id); END IF; IF item.duration_days > drugs.max_duration_days THEN INSERT INTO violations VALUES ('EXCESSIVE_DURATION','LOW',rx_id); END IF; END FOR; SET score = (passed_checks / total_checks) * 100; INSERT INTO compliance_reports VALUES (rx_id, score, IF(score>=75,'PASS','FAIL')); END;
VIEW: Doctor Compliance Summary
CREATE VIEW doctor_compliance_view AS SELECT doctor_name, AVG(compliance_score) AS avg_score, COUNT(*) AS total_prescriptions, SUM(total_violations) AS total_violations, CASE WHEN AVG(compliance_score) >= 90 THEN 'EXCELLENT' WHEN AVG(compliance_score) >= 75 THEN 'GOOD' ELSE 'NEEDS REVIEW' END AS rating FROM compliance_reports GROUP BY doctor_name ORDER BY avg_score DESC;
